Google Apps Script Code

Copy code ini ke Google Apps Script Editor. (Sudah diperbarui dengan Upsert Logic agar data aman multi-device):

// Google Apps Script untuk E-Tamu
function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const rows = data.slice(1);
  
  const guests = rows.map(row => {
    const guest = {};
    headers.forEach((header, i) => { guest[header] = row[i]; });
    return guest;
  });
  
  return ContentService.createTextOutput(JSON.stringify({
    success: true, data: guests, sheetUrl: SpreadsheetApp.getActiveSpreadsheet().getUrl()
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    const data = JSON.parse(e.postData.contents);
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(['id', 'name', 'agency', 'purpose', 'phone', 'date', 'time', 'signature', 'syncedAt']);
    }
    
    if (data.action === 'create') {
      sheet.appendRow([
        data.guest.id, data.guest.name, data.guest.agency, data.guest.purpose,
        data.guest.phone, data.guest.date, data.guest.time, data.guest.signature || '', new Date().toISOString()
      ]);
      return ContentService.createTextOutput(JSON.stringify({ success: true, message: 'Data disimpan', sheetUrl: SpreadsheetApp.getActiveSpreadsheet().getUrl() })).setMimeType(ContentService.MimeType.JSON);
      
    } else if (data.action === 'delete') {
      const values = sheet.getDataRange().getValues();
      for (let i = values.length - 1; i >= 1; i--) {
        if (values[i][0] == data.id) { sheet.deleteRow(i + 1); break; }
      }
      return ContentService.createTextOutput(JSON.stringify({ success: true, message: 'Data dihapus' })).setMimeType(ContentService.MimeType.JSON);
      
    } else if (data.action === 'sync_all') {
      const existingIds = sheet.getDataRange().getValues().slice(1).map(row => String(row[0])); 
      let added = 0;
      data.guests.forEach(guest => {
        if (!existingIds.includes(String(guest.id))) {
          sheet.appendRow([
            guest.id, guest.name, guest.agency, guest.purpose, guest.phone, guest.date, guest.time, guest.signature || '', new Date().toISOString()
          ]);
          added++;
        }
      });
      return ContentService.createTextOutput(JSON.stringify({ success: true, message: added + ' data disync', sheetUrl: SpreadsheetApp.getActiveSpreadsheet().getUrl() })).setMimeType(ContentService.MimeType.JSON);
    }
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
