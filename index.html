// --- SISTEM EXPORT (DENGAN TANDA TANGAN) --- function exportExcel() { if(!state.guests.length) return showToast("Data Kosong", "error"); // CSV tidak bisa menyimpan gambar, jadi kita beri status teks let csv = "ID,TANGGAL,WAKTU,NAMA,INSTANSI,NO_HP,KEPERLUAN,TANDA_TANGAN\n"; state.guests.forEach(g => csv += `${g.id},${g.date},${g.time},"${g.name}","${g.agency}","${g.phone}","${g.purpose}","Terekam"\n`); saveAs(new Blob([csv], {type:'text/csv;charset=utf-8;'}), `Data_Tamu_SMK_${new Date().getTime()}.csv`); showToast("Excel Berhasil Diunduh"); } // Fungsi Bantuan untuk mengubah string Tanda Tangan menjadi file Gambar yang bisa dibaca Word function base64ToUint8Array(base64) { const base64String = base64.split(',')[1]; const binaryString = window.atob(base64String); const len = binaryString.length; const bytes = new Uint8Array(len); for (let i = 0; i < len; i++) { bytes[i] = binaryString.charCodeAt(i); } return bytes; } async function exportWord() { if(!state.guests.length) return showToast("Data Kosong", "error"); toggleLoader(true, "Menyusun Docx..."); try { // Perhatikan penambahan ImageRun disini const { Document, Packer, Paragraph, Table, TableRow, TableCell, WidthType, TextRun, ImageRun, AlignmentType } = docx; // Header Tabel const rows = [ new TableRow({ children: [ new TableCell({children:[new Paragraph({children:[new TextRun({text:"TGL/WAKTU", bold:true})]})]}), new TableCell({children:[new Paragraph({children:[new TextRun({text:"NAMA", bold:true})]})]}), new TableCell({children:[new Paragraph({children:[new TextRun({text:"INSTANSI", bold:true})]})]}), new TableCell({children:[new Paragraph({children:[new TextRun({text:"KEPERLUAN", bold:true})]})]}), new TableCell({children:[new Paragraph({children:[new TextRun({text:"TANDA TANGAN", bold:true})]})]}) ]}) ]; // Isi Data Tabel state.guests.forEach(g => { let signatureRender; // Proses Tanda Tangan menjadi Gambar Word if (g.signature && g.signature.includes('base64')) { try { const imageBuffer = base64ToUint8Array(g.signature); signatureRender = new Paragraph({ children: [ new ImageRun({ data: imageBuffer, transformation: { width: 80, height: 40 }, // Ukuran gambar di dalam tabel Word type: "png" }) ], alignment: AlignmentType.CENTER }); } catch (err) { signatureRender = new Paragraph("Error TTD"); } } else { signatureRender = new Paragraph("-"); } rows.push(new TableRow({ children: [ new TableCell({children:[new Paragraph(`${g.date}\n${g.time}`)]}), new TableCell({children:[new Paragraph(g.name)]}), new TableCell({children:[new Paragraph(g.agency)]}), new TableCell({children:[new Paragraph(g.purpose)]}), new TableCell({children:[signatureRender]}) // Gambar dimasukkan kesini ]})); }); const doc = new Document({ sections: [{ children: [new Paragraph({text:"LAPORAN BUKU TAMU SMK QUEEN AL-FALAH", heading:"Heading1"}), new Paragraph({text:"\n"}), new Table({width:{size:100, type:WidthType.PERCENTAGE}, rows})] }] }); const blob = await Packer.toBlob(doc); saveAs(blob, `Laporan_Tamu_${new Date().getTime()}.docx`); showToast("Word Berhasil Diunduh"); } catch(e) { console.error(e); showToast("Gagal Buat File", "error"); } finally { toggleLoader(false); } }
