info Cara Setup:

  1. Buat Google Sheets baru

  3. Copy script yang disediakan
  4. Deploy → New deployment → Web app
  5. PENTING: Set "Who has access" ke "Anyone"
  6. Copy URL Web App dan paste di atas

Google Apps Script Code

Copy code ini ke Google Apps Script Editor. (Sudah diperbarui dengan Upsert Logic agar data aman multi-device):

// Google Apps Script untuk E-Tamu
function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  
  // Baca semua data
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const rows = data.slice(1);
  
  const guests = rows.map(row => {
    const guest = {};
    headers.forEach((header, i) => {
      guest[header] = row[i];
    });
    return guest;
  });
  
  return ContentService.createTextOutput(JSON.stringify({
    success: true,
    data: guests,
    // Mengembalikan URL spreadsheet agar bisa dibuka di aplikasi
    sheetUrl: SpreadsheetApp.getActiveSpreadsheet().getUrl()
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    // Parse contents even if sent as text/plain
    const data = JSON.parse(e.postData.contents);
    
    // Setup headers jika sheet kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(['id', 'name', 'agency', 'purpose', 'phone', 'date', 'time', 'signature', 'syncedAt']);
    }
    
    if (data.action === 'create') {
      // Tambah data baru
      sheet.appendRow([
        data.guest.id,
        data.guest.name,
        data.guest.agency,
        data.guest.purpose,
        data.guest.phone,
        data.guest.date,
        data.guest.time,
        data.guest.signature || '',
        new Date().toISOString()
      ]);
      
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        message: 'Data berhasil disimpan',
        sheetUrl: SpreadsheetApp.getActiveSpreadsheet().getUrl()
      })).setMimeType(ContentService.MimeType.JSON);
      
    } else if (data.action === 'delete') {
      // Hapus data
      const dataRange = sheet.getDataRange();
      const values = dataRange.getValues();
      
      for (let i = values.length - 1; i >= 1; i--) {
        if (values[i][0] == data.id) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
      
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        message: 'Data berhasil dihapus'
      })).setMimeType(ContentService.MimeType.JSON);
      
    } else if (data.action === 'sync_all') {
      // PERBAIKAN: Jangan hapus semua data! Lakukan pencocokan ID (Upsert) untuk mencegah bentrok multi-device
      const dataRange = sheet.getDataRange();
      const existingData = dataRange.getValues();
      const existingIds = existingData.slice(1).map(row => String(row[0])); 
      
      let added = 0;
      data.guests.forEach(guest => {
        // Jika ID belum ada di database, tambahkan sebagai baris baru
        if (!existingIds.includes(String(guest.id))) {
          sheet.appendRow([
            guest.id,
            guest.name,
            guest.agency,
            guest.purpose,
            guest.phone,
            guest.date,
            guest.time,
            guest.signature || '',
            new Date().toISOString()
          ]);
          added++;
        }
      });
      
      return ContentService.createTextOutput(JSON.stringify({
        success: true,
        message: `${added} data baru berhasil ditambahkan dari hasil sync`,
        sheetUrl: SpreadsheetApp.getActiveSpreadsheet().getUrl()
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
